cmake_minimum_required(VERSION 3.20)

project(prosperobrowser LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

if(CMAKE_CROSSCOMPILING)
  set(_prosperobrowser_tests_default OFF)
else()
  set(_prosperobrowser_tests_default ON)
endif()
option(PROSPEROBROWSER_BUILD_TESTS "Build unit tests" ${_prosperobrowser_tests_default})

set(PROSPEROBROWSER_FRONTEND "none" CACHE STRING "Frontend to build: none, host-gtk or ps5")
set_property(CACHE PROSPEROBROWSER_FRONTEND PROPERTY STRINGS none host-gtk ps5)
if(NOT PROSPEROBROWSER_FRONTEND MATCHES "^(none|host-gtk|ps5)$")
  message(FATAL_ERROR "PROSPEROBROWSER_FRONTEND must be one of: none, host-gtk, ps5 (got '${PROSPEROBROWSER_FRONTEND}')")
endif()

set(PROSPEROBROWSER_THIRD_PARTY_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party")

add_library(prosperobrowser_warnings INTERFACE)
target_compile_options(prosperobrowser_warnings INTERFACE -Wall -Wextra -Werror)

add_subdirectory(src/core)
add_subdirectory(src/shell)
add_subdirectory(src/ps5/spawn)
add_subdirectory(src/ps5/audio)
add_subdirectory(src/ps5/display)
add_subdirectory(src/ps5/video)
add_subdirectory(src/ps5/cpu)
add_subdirectory(src/chrome-ui)

if(PROSPEROBROWSER_FRONTEND STREQUAL "host-gtk")
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/frontend/host-gtk/CMakeLists.txt")
    add_subdirectory(src/frontend/host-gtk)
  else()
    message(FATAL_ERROR "host-gtk frontend is not available yet")
  endif()
elseif(PROSPEROBROWSER_FRONTEND STREQUAL "ps5")
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/frontend/ps5/CMakeLists.txt")
    add_subdirectory(src/frontend/ps5)
  else()
    message(FATAL_ERROR "ps5 frontend is not available yet")
  endif()
endif()

if(PROSPEROBROWSER_BUILD_TESTS)
  enable_testing()
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/core/CMakeLists.txt")
    add_subdirectory(tests/core)
  endif()
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/shell/CMakeLists.txt")
    add_subdirectory(tests/shell)
  endif()
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/ps5/CMakeLists.txt")
    add_subdirectory(tests/ps5)
  endif()
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/media/CMakeLists.txt")
    add_subdirectory(tests/media)
  endif()
endif()
